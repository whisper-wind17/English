name: Execute Third-party Learner Review Approval

on:
  push:
    branches: [main]
    paths:
      - '.github/workflows/approve-third-party-learner-review.yml'

permissions:
  contents: write

jobs:
  approve:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Confirm current review scope is build-valid before approval
        run: |
          python tools/check_third_party_learner_content.py
          python tools/validate_third_party_pronunciation_materialization.py
          python tools/check_klose_learner.py
      - name: Approve current pending learner presentations
        run: |
          python tools/approve_klose_learner_review.py \
            --batch-id third-party-level4-review-20260912 \
            --profile klose \
            --level 4 \
            --expected-count 2059 \
            --reviewer-type model \
            --review-note 'Independent risk-stratified model review: 1821 new presentations passed full content gates and semantic high-risk adjudication; 160 reuse guardrails preserved existing presentation; 78 released invalidations re-reviewed; 616 pronunciation rows independently reviewed; 16 explicit content corrections applied; approval bound to current fingerprints.' \
            --confirm-all-current \
            --pending-only
      - name: Verify approval closed required review scope
        run: |
          python tools/sync_klose_learner_review_registry.py
          python - <<'PY'
          import csv
          from pathlib import Path
          base=Path('anki/klose')
          def read(path):
              with path.open('r',encoding='utf-8-sig',newline='') as f:
                  return list(csv.DictReader(f))
          master=read(base/'master/vocabulary_master.csv')
          learner={r['NoteID']:r for r in read(base/'learner/current.csv')}
          admission=read(base/'learner/learning_admission.csv')
          registry=read(base/'learner/presentation_review_registry.csv')
          released={r['NoteID'] for r in master if r.get('Released')=='yes' and r['NoteID'] in learner and learner[r['NoteID']].get('LearnerProfile')=='klose' and learner[r['NoteID']].get('LearnerLevel')=='4'}
          allowed={r['NoteID'] for r in admission if r.get('LearnerProfile')=='klose' and r.get('LearnerLevel')=='4' and r.get('Status')=='allowed'}
          required=released|allowed
          by={(r.get('LearnerProfile'),r.get('LearnerLevel'),r.get('NoteID')):r for r in registry}
          required_rows=[by.get(('klose','4',nid)) for nid in sorted(required)]
          missing=[nid for nid,row in zip(sorted(required),required_rows) if row is None]
          pending=[row['NoteID'] for row in required_rows if row is not None and row.get('ReviewStatus')=='pending']
          unreviewed=[row['NoteID'] for row in required_rows if row is not None and row.get('ReviewStatus') not in {'model-reviewed','human-reviewed'}]
          historical=[r for r in registry if r.get('LearnerProfile')=='klose' and r.get('LearnerLevel')=='4' and r.get('NoteID') not in required]
          if len(required)!=2953 or missing or pending or unreviewed:
              raise SystemExit(f'Approval closure failed: required={len(required)} missing={len(missing)} pending={len(pending)} unreviewed={len(unreviewed)}')
          print(f'Approval closure PASS: required={len(required)} pending=0 historical_out_of_scope={len(historical)}')
          PY
      - name: Commit immutable approval manifest and review registry
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+whisper-wind17@users.noreply.github.com'
          git add \
            anki/klose/learner/presentation_review_registry.csv \
            anki/klose/learner/review_approvals/third-party-level4-review-20260912.csv \
            anki/klose/master/build_stats.csv
          changed="$(git diff --cached --name-only | sort)"
          expected="$(printf '%s\n' \
            'anki/klose/learner/presentation_review_registry.csv' \
            'anki/klose/learner/review_approvals/third-party-level4-review-20260912.csv' \
            'anki/klose/master/build_stats.csv' | sort)"
          if [ "$changed" != "$expected" ]; then
            echo 'Unexpected approval diff:'; echo "$changed"; exit 1
          fi
          git commit -m 'review: approve third-party LearnerLevel 4 presentations'
          git pull --rebase origin main
          git push origin HEAD:main
